const auth = firebase.auth(); const db = firebase.firestore(); let currentUser = null; let staff = null; async function login() { const msg = document.getElementById("message"); try { msg.innerText = "Logging in..."; const email = document.getElementById("email").value.trim(); const password = document.getElementById("password").value; const result = await auth.signInWithEmailAndPassword(email, password); const uid = result.user.uid; const doc = await db.collection("staff").doc(uid).get(); if (!doc.exists) { msg.innerText = "Staff profile not found. UID: " + uid; await auth.signOut(); return; } staff = doc.data(); currentUser = result.user; msg.innerText = ""; showApp(); } catch (e) { msg.innerText = "Error: " + e.message; console.error(e); } }
